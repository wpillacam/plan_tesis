\section*{Símbolos}
\begin{longtable}{@{}p{2.2cm}p{12.5cm}@{}}
$A$ & Matriz de comparación pareada del Proceso Analítico Jerárquico (AHP) \\
$a_{ij}$ & Elemento de la matriz $A$: juicio de importancia relativa del criterio $i$ respecto al $j$ \\
$CR$ & Razón de Consistencia (\emph{Consistency Ratio}); aceptable si $CR \leq 0.10$ \\
$\lambda_{\text{máx}}$ & Autovalor principal (máximo) de la matriz de comparación pareada $A$ \\
$n$ & Número total de criterios de evaluación considerados en el modelo \\
$r_{fi}$ & Calificación del framework $f$ en el criterio $i$, en escala $[0,10]$ \\
$RI_n$ & Índice de Consistencia Aleatorio tabulado para matrices de orden $n$ \\
$S_f$ & Puntuación ponderada global del framework $f$: $S_f = \sum_{i=1}^{n} w_i \cdot r_{fi}$ \\
$w$ & Vector de prioridades (pesos normalizados) derivado del proceso AHP \\
$w_i$ & Peso ponderado del criterio $i$; cumple $\sum_{i=1}^{n} w_i = 1$, $w_i \geq 0$ \\
\end{longtable}
